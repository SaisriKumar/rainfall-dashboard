from flask import Flask, render_template, request, jsonify, send_from_directory import pdfplumber import os import re import pandas as pd from datetime import datetime, timedelta app = Flask(__name__) # --- PATH CONFIGURATION --- BASE_DIR = r"D:\IMD_Data\IMD_Rainfall" PDF_DIR = os.path.join(BASE_DIR, "PDF") TEMPLATES_DIR = os.path.join(BASE_DIR, "templates") CSV_PATH = os.path.join(TEMPLATES_DIR, "Search_list.csv") GEOJSON_PATH = os.path.join(TEMPLATES_DIR, "India_Districts.geojson") os.makedirs(PDF_DIR, exist_ok=True) def parse_date(date_str): """Parses input date strings (HTML standard is YYYY-MM-DD).""" for fmt in ('%Y-%m-%d', '%d-%m-%Y'): try: return datetime.strptime(date_str, fmt) except ValueError: continue return None def get_district_list(): """Reads Search_list.csv and maps Districts to their parent State/UT.""" if not os.path.exists(CSV_PATH): return [] df = pd.read_csv(CSV_PATH) districts = [] current_state = "Unknown" for _, row in df.iterrows(): name = str(row['Select District']).strip() category = str(row['List']).strip() if category == "State/UT": current_state = name elif category == "District": districts.append(f"{name} ({current_state})") return sorted(districts) @app.route('/') def index(): return render_template('index.html', districts=get_district_list()) @app.route('/get_map_data') def get_map_data(): """Serves the India_Districts.geojson file for the Leaflet map.""" return send_from_directory(TEMPLATES_DIR, "India_Districts.geojson") @app.route('/search_range', methods=['POST']) def search_range(): data = request.json start_dt = parse_date(data.get('startDate')) end_dt = parse_date(data.get('endDate')) full_dist_name = data.get('district') if not start_dt or not end_dt or not full_dist_name: return jsonify({"error": "Missing input data"}), 400 # Extract district name (e.g., "NICOBAR" from "NICOBAR (A & N ISLAND)") clean_dist = full_dist_name.split(" (")[0].strip() results = [] # IMD Table Regex Pattern pattern = rf"{clean_dist}\s+([\d\.\-]+)\s+([\d\.\-]+)\s+([\d\.\-%]+)\s+(\w+)\s+([\d\.\-]+)\s+([\d\.\-]+)\s+([\d\.\-%]+)\s+(\w+)" curr = start_dt while curr <= end_dt: # NEW FILENAME FORMAT: YYYY-MM-DD date_str = curr.strftime('%Y-%m-%d') fname = f"{date_str}_DISTRICT_RAINFALL_DISTRIBUTION_COUNTRY_INDIA_cd.pdf" fpath = os.path.join(PDF_DIR, fname) row_vals = None if os.path.exists(fpath): try: with pdfplumber.open(fpath) as pdf: for page in pdf.pages: text = page.extract_text() match = re.search(pattern, text, re.IGNORECASE) if text else None if match: row_vals = list(match.groups()) break except Exception as e: print(f"Error reading {fname}: {e}") results.append({ "date": date_str, "found": bool(row_vals), "data": row_vals }) curr += timedelta(days=1) return jsonify({"results": results}) if __name__ == '__main__': app.run(debug=True)
